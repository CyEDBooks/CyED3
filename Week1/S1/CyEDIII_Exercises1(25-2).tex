\section{{Regular languages and automata theory}}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%



\begin{ejer}
Given $\Sigma = \{a,b,c, d\}$ define $\Sigma^*$
\end{ejer}


\begin{ejer}
Given $u \pertenece \Sigma^*$ and $n \pertenece \mathbb{N},$ give a recursive definition for $u^n.$
\end{ejer}


\begin{ejer}
Given $u \pertenece \Sigma^*$ and $a \pertenece \Sigma$ give a recursive definition for $|u|.$
\end{ejer}


\begin{ejer}
Given $u \pertenece \Sigma^*$ and $a \pertenece \Sigma$ give a recursive definiton for $u^R.$
\end{ejer}


\begin{ejer}
If $u,v \pertenece \Sigma^*,$ show that $(u v)^R = v^R u^R.$ 
\end{ejer}


\begin{ejer}
Given an example of an alphabet $\Sigma$ and two different languages $A, B$ over $\Sigma$ such that $AB = BA.$
\end{ejer}

\begin{ejer}
Show that $A \cdot (B \cdot C) = (A \cdot B) \cdot C.$
\end{ejer}


\begin{ejer}
Show that $A \cdot \{\lambda\} = \{\lambda\} \cdot A = A.$
\end{ejer}


\begin{ejer}
Show that $A \cdot \emptyset = \emptyset \cdot A = \emptyset.$
\end{ejer}
