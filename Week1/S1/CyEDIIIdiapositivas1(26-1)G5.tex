\section[Regular languages and Automata theory]{Regular languages and Automata theory}
\subsection{Alphabets, strings and languages}
\begin{frame}{Alphabets, strings and languages}
What is an alphabet? \pause
\begin{df}
An alphabet is a finite, noneempty set of symbols. We use the symbol $\Sigma$ to define it. \pause
\end{df}
Which are some examples of alphabets? \pause
\begin{block}{}
\begin{itemize}
\item The binary alphabet: $\Sigma = \{0,1\}$
\item The set of all lower case vowels: $\Sigma = \{a,e,i,o,u\}.$ 
\item The set of all ASCII characters, or the set of all printable ASCII characters.
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Alphabets, strings and languages}
What is a string? \pause
\begin{df}
A string or a word is a finite sequence of symbols chosen from some alphabet. \pause
\end{df}
Which are some examples of strings? \pause
\begin{block}{}
\begin{itemize}
\item $01101$ is a string from the binary alphabet $\Sigma = \{0,1\}$
\item $banana$ is a string from the all lower case letters` alphabet $\Sigma = \{a,b,c,...,z\}.$ 
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Alphabets, strings and languages}
\begin{ej}
Let $\Sigma = \{a,b\}$. \\ \pause Here we have strings from $\Sigma:$ 
\begin{center}
\begin{tabular}{l}
$aba$ \\ 
$ababaaa$ \\ 
$aaaab$ \pause
\end{tabular}
\end{center}
We can see that $aba \neq aab,$ due to the fact that the order of symbols is important when dealing with strings, since strings work as sequences. 
\end{ej}
\end{frame}

\begin{frame}{Alphabets, strings and languages}
What is the empty string? \pause
\begin{df}
The empty string is the string with zero occurrences of symbols. This string is denoted by $\lambda$ or $\epsilon.$ This string may be chosen from any alphabet whatsoever. 
\end{df}
\end{frame}

\begin{frame}{Alphabets, strings and languages}
What about the length of a string? \pause
\begin{block}{}
\begin{itemize}
\item It is useful to classify strings by their length.
\item It is common to say that the length of a string is the number of symbols in the string.
\item Not strictly correct. $10101$ has only two symbols but there are five positions for symbols (its length is 5).
\item Generally expect that the number of symbols can be used when. number of positions is meant.
\item The standard notation for the length of a string $w$ is $|w|.$
\end{itemize}
\end{block}
\end{frame}

\begin{frame}{Alphabets, strings and languages}
How could we formally define the length of a string? \pause
\begin{df}
\begin{center}
\[|u|=\left\{
\begin{array}{c l}      
    0 & \text{if } u = \lambda \\
    n & \text{if } u = a_1 a_2 \cdot \cdot \cdot a_n.
\end{array}\right.\] \pause
\end{center}
\end{df}
\begin{ej}
$|aba| = 3, \, |baaa| = 4, \, |\lambda| = 0.$
\end{ej}
\end{frame}

\begin{frame}{Alphabets, strings and languages}
What are the powers of an alphabet? \pause
\begin{block}{}
\begin{itemize}
\item Let $\Sigma$ be an alphabet. 
\item We can express the set of all strings of a certain length from that alphabet by using an exponential notation.
\item We define $\Sigma^k$ to be the set of strings of length $k,$ each of whose symbols is in $\Sigma.$ \pause
\end{itemize}
\end{block}
\begin{ej}
\begin{itemize}
\item $\Sigma^0 = \{\lambda\}$ regardless of what alphabet $\Sigma$ is.
\item If $\Sigma = \{0,1\},$ then $\Sigma^1 = \{0,1\}, \, \Sigma^2 = \{00,01,10,11\}.$ 
\end{itemize}
\end{ej}
\end{frame}

\begin{frame}{Alphabets, strings and languages}
What is the Kleene star of an alphabet? \pause
\begin{block}{}
\begin{itemize}
\item The set of all possible strings over an alphabet $\Sigma.$
\item It is denoted as $\Sigma^*.$
\item $\Sigma^* = \Sigma^0 \, \cup \, \Sigma^1 \, \cup \, \Sigma^2 \, \cdot \cdot \cdot$ \pause
\end{itemize}
\end{block}
\begin{ej}
\begin{itemize}
\item If $\Sigma = \{0,1\},$ then $\Sigma^* = \{\lambda,0,1,00,01,10,11,000,...\}.$ 
\end{itemize}
\end{ej}
\end{frame}

\begin{frame}{Alphabets, strings and languages}
What is the positive closure of an alphabet? \pause
\begin{block}{}
\begin{itemize}
\item The set of all possible strings over an alphabet $\Sigma$ without the empty string.
\item It is denoted as $\Sigma^+.$
\item $\Sigma^+ = \Sigma^1 \, \cup \, \Sigma^2 \, \cup \, \Sigma^3 \, \cup \, \cdot \cdot \cdot$ \pause
\end{itemize}
\end{block}
\begin{ej}
\begin{itemize}
\item If $\Sigma = \{0,1\},$ then $\Sigma^* = \{0,1,00,01,10,11,000,...\}.$ 
\end{itemize}
\end{ej}
\end{frame}

\begin{frame}{Alphabets, strings and languages}
What is string concatenation? \pause
\begin{df}
Given an alphabet $\Sigma$ and two strings $u,v \pertenece \Sigma^*,$ the concatenation of $u$ and $v$ is expressed as $u \cdot v$ or simply $u v.$  \pause We can define it as  \pause If $u,v \pertenece \Sigma^*, \, a \pertenece \Sigma,$ then \pause
\begin{enumerate}
\item $u \cdot \lambda = \lambda \cdot u = u$ \pause
\item $u \cdot (v a) = (u \cdot v) a$
\end{enumerate}
\end{df}
\end{frame}

\begin{frame}{Alphabets, strings and languages}
What is the reverse or inverse of a string? \pause
\begin{df}
The reverse or inverse of a string $u \pertenece \Sigma^*$ is expressed as $u^R$ and defined as: \pause
\begin{center}
\[u^R=\left\{
\begin{array}{c l}      
    \lambda & \text{if } u = \lambda \\
    a_n \cdot \cdot \cdot a_2 a_1 & \text{if } u = a_1 a_2 \cdot \cdot \cdot a_n.
\end{array}\right.\] 
\end{center}
\end{df}
\end{frame}

{\footnotesize \begin{frame}{Alphabets, strings and languages}
What is a substring? \pause
\begin{df}
A string $v$ is a substring or subword of $u$ if there exist strings $x, \, y$ such that $u = xvy.$ Let's point out that $x$ or $y$ might be $\lambda$ and therefore, the empty string is a substring of any string and any string is a substring of itself. \pause
\end{df}
What is a prefix? \pause
\begin{df}
A prefix of $u$ is a string $v$ such that $u = v w$ for any string $w \pertenece \Sigma^*.$ We say that $v$ is a proper prefix if $v \neq u.$ \pause
\end{df}
What is a suffix? \pause
\begin{df}
A suffix of $u$ is a string $v$ such that $u = w v$ for any string $w \pertenece \Sigma^*.$ We say that $v$ is a proper suffix if $v \neq u.$ 
\end{df}
\end{frame}}

{\footnotesize \begin{frame}{Alphabets, strings and languages}
See that $\lambda$ is a prefix and suffix of $u$ since $u \lambda = \lambda u = u.$ By the same reasoning $u$ is a prefix and suffix of itself. \pause
\begin{ej}
Given $\Sigma = \{a,d,c,d\}$ and $u = bcbaadb.$
\begin{center}
\begin{tabular}{rr}
Prefixes of $u:$ & Suffixes of $u:$ \\
$\lambda$ & $\lambda$ \\
$b$ & $b$ \\
$bc$ & $db$ \\
$bcb$ & $adb$ \\
$bcba$ & $aadb$ \\
$bcbaa$ & $baadb$ \\
$bcbaaad$ & $cbaadb$ \\
$bcbaadb$ & $bcbaadb$ 
\end{tabular}
\end{center}
\end{ej}
\end{frame}}

\begin{frame}{Alphabets, strings and languages}
What is a language? \pause
\begin{df}
A language $L$ over an alphabet $\Sigma$ is a subset of $\Sigma^*,$ that is $L \subseteq \Sigma^*$ \\ \pause
Border cases:
\begin{center}
\begin{tabular}{ll}
$L = \emptyset,$ & the empty language. \\
$L = \Sigma^*,$ & the language of every string over $\Sigma.$ \pause
\end{tabular}
\end{center}
\end{df}
Any language $L$ fulfills $\emptyset \subseteq L \subseteq \Sigma^*,$ and it can be either finite or infinite. We represent languages with upper case letters $A, \, B, \, C,...,L, \, M, \, N,...$
\end{frame}
%%%%%%%%%%%%%
\begin{frame}{Alphabets, strings and languages}
\begin{ej}
The following are examples of languages over given alphabets. \pause
\begin{itemize}
\item $\Sigma = \{a,b,c\}.$ $L=\{a,aba,aca\}.$ \pause
\item $\Sigma = \{a,b,c\}.$ $L=\{a,aa,aaa,...\} = \{a^n \talque n \geq 1\}.$ \pause
\item $\Sigma = \{a,b,c\}.$ $L=\{\lambda,aa,aba,ab^2a,ab^3a,...\}=$ $\{ab^na \talque n \geq 0\} \cup \{\lambda\}.$ 
\end{itemize}
\end{ej}
\end{frame}

{\footnotesize\begin{frame}{Alphabets, strings and languages}
Which are the operations between languages? \pause
\begin{df}
Since languages over $\Sigma$ are subsets of $\Sigma^*,$ the usual operations between sets are also valid operations between languages. Thení, if $A$ and $B$ are languages over $\Sigma$ ($A, \, B \subseteq \Sigma^*$), then the following are also languages over $\Sigma:$
\begin{center}
\begin{tabular}{ll}
$A \cup B$ & Union \pause \\
$A \cap B$ & Intersection \pause \\
$A-B$ & Diference \pause \\
$\overline{A} = \Sigma^*-A$ & Complement \pause
\end{tabular}
\end{center}
\end{df}
These operations between languages are called boolean operations in order to differentiate them from the linguistic operations, which are extensions to the languages of the operations between strings.
\end{frame}}

\begin{frame}{Alphabets, strings and languages}
What is the concatenation between languages? \pause
\begin{df}
The concatenation of two languages $A$ and $B$ over $\Sigma,$ represented as $A \cdot B$ or simply $A B$ is defined as: \pause
\[A B = \{u v \talque u \pertenece A, \, v \pertenece B\}\] \pause
In general, $A B \neq B A$
\end{df}
\end{frame}

{\scriptsize \begin{frame}{Alphabets, strings and languages}
Which are the properties of concatenation of languages? \pause
\begin{df}
Given languages $A, B, C$ over $\Sigma,$ i.e.  $A, \, B, \, C \subseteq \Sigma^*.$ Then \pause
\begin{enumerate}
\item $A \cdot \emptyset = \emptyset \cdot A = \emptyset.$ \pause
\item $A \cdot \{\lambda\} = \{\lambda\} \cdot A = A.$ \pause
\item Associative property,
\begin{center}
$A \cdot (B \cdot C) = (A \cdot B) \cdot C.$
\end{center} \pause
\item Distributivity of concatenation with respect to union,
\begin{center}
$A \cdot (B \cup C) = A \cdot B \cup A \cdot C.$ \\
$(B \cup C) \cdot A = B \cdot A \cup C \cdot A.$ 
\end{center} \pause
\item Generalized distributivity property. If $\{B_i\}_{i \pertenece I}$ is any family of languages over $\Sigma,$ then
\begin{center}
$A \cdot \bigcup\limits_{i \pertenece I} B_i = \bigcup\limits_{i \pertenece I} (A \cdot B_i),$ \\
$\Big( \bigcup\limits_{i \pertenece I} B_i\Big) \cdot A = \bigcup\limits_{i \pertenece I} (B_i \cdot A).$ 
\end{center} 
\end{enumerate}
\end{df}
\end{frame}}
